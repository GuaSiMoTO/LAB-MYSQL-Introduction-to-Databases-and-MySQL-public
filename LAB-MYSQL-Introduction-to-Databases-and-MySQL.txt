-- 3.1 Create in MySQLWorkbench a database called Rebel_Alliance
CREATE DATABASE rebel_alliance;

-- 3.2 Selecting the Database
USE rebel_alliance;

-- 3.3 Add Previous tables : Planets, Ships, Characters

CREATE TABLE Planets (
    planet_id INT PRIMARY KEY,
    planet_name VARCHAR(50),
    region VARCHAR(50),
    population BIGINT,
    discovered_date DATE
);

CREATE TABLE Ships (
    ship_id INT PRIMARY KEY,
    ship_name VARCHAR(50),
    ship_type ENUM('Freighter', 'Starfighter', 'Capital Ship'),
    capacity INT,
    in_service SET('Rebel Alliance', 'Galactic Empire', 'Independent')
);

CREATE TABLE Characters (
    character_id INT PRIMARY KEY,
    character_name VARCHAR(50),
    affiliation VARCHAR(50),
    home_planet VARCHAR(50),
    birth_date VARCHAR(10)
);

-- 3.4 Add these two new tables (Missions, Commanders)

CREATE TABLE Missions (
    mission_id int primary key,
    mission_name varchar(50),
    start_date date,
    end_date date,
    status_mission enum('Planned','Ongoing','Completed')
);

CREATE TABLE Commanders (
    commander_id int primary key,
    commander_name varchar(50),
    status_commander varchar(50),
    assigned_mission int
);
